def get_referenced_annotations(query):
    """
    Determine which annotations are actually referenced by the query.
    
    An annotation is referenced if it appears in:
    - Filter expressions (WHERE clause)
    - Order by expressions
    - Other aggregates that reference it
    """
    referenced = set()
    
    # Check WHERE clause
    def check_where(node):
        if hasattr(node, 'children'):
            for child in node.children:
                result = check_expr(child)
                if result:
                    return True
        return False
    
    def check_expr(expr):
        if expr is None:
            return False
        if hasattr(expr, 'get_source_expressions'):
            for source in expr.get_source_expressions():
                if hasattr(source, 'name') and source.name in query.annotation_select:
                    referenced.add(source.name)
                    return True
                if hasattr(source, 'get_source_expressions'):
                    if check_expr(source):
                        return True
        elif hasattr(expr, 'children'):
            return check_where(expr)
        return False
    
    # Check WHERE clause
    if hasattr(query, 'where') and query.where:
        check_where(query.where)
    
    # Check ORDER BY
    for field in query.order_by:
        if hasattr(field, 'expression') and hasattr(field.expression, 'name'):
            if field.expression.name in query.annotation_select:
                referenced.add(field.expression.name)
        elif hasattr(field, 'expression') and hasattr(field.expression, 'get_source_expressions'):
            for source in field.expression.get_source_expressions():
                if hasattr(source, 'name') and source.name in query.annotation_select:
                    referenced.add(source.name)
                    break
    
    # Check extra_order_by
    for field in query.extra_order_by:
        if hasattr(field, 'expression') and hasattr(field.expression, 'name'):
            if field.expression.name in query.annotation_select:
                referenced.add(field.expression.name)
        elif hasattr(field, 'expression') and hasattr(field.expression, 'get_source_expressions'):
            for source in field.expression.get_source_expressions():
                if hasattr(source, 'name') and source.name in query.annotation_select:
                    referenced.add(source.name)
                    break
    
    # Check combined queries
    for cq in getattr(query, 'combined_queries', []):
        referenced.update(get_referenced_annotations(cq))
    
    return referenced
