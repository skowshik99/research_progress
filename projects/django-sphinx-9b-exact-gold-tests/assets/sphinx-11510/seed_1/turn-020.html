class Include(BaseInclude, SphinxDirective):
    """
    Like the standard "Include" directive, but interprets absolute paths
    "correctly", i.e. relative to source directory.
    """

    def run(self) -> list[Node]:
        if self.arguments[0].startswith('<') and \
           self.arguments[0].endswith('>'):
            # docutils "standard" includes, do not do path processing
            return super().run()
        
        rel_filename, filename = self.env.relfn2path(self.arguments[0])
        self.arguments[0] = filename
        self.env.note_included(filename)
        
        # If the include will be parsed as RST (not literal), we need to
        # trigger the source-read event for the included file
        if 'literal' not in self.options:
            # Read the file content first
            with open(filename, encoding=self.state.document.settings.input_encoding,
                      errors=self.state.document.settings.input_encoding_error_handler) as f:
                raw_content = f.read()
            
            # Apply line clipping if specified
            startline = self.options.get('start-line', None)
            endline = self.options.get('end-line', None)
            after_text = self.options.get('start-after', None)
            before_text = self.options.get('end-before', None)
            
            if startline or endline:
                lines = raw_content.splitlines(keepends=True)
                if startline:
                    lines = lines[startline:]
                if endline:
                    lines = lines[:endline]
                raw_content = ''.join(lines)
            
            if after_text:
                idx = raw_content.find(after_text)
                if idx >= 0:
                    raw_content = raw_content[idx + len(after_text):]
            
            if before_text:
                idx = raw_content.find(before_text)
                if idx >= 0:
                    raw_content = raw_content[:idx]
            
            # Trigger source-read event with the included file's docname
            arg = [raw_content]
            self.env.events.emit('source-read', filename, arg)
            raw_content = arg[0]
        
        self.state.document.settings.record_dependencies.add(filename)
        return super().run()
