class Include(BaseInclude, SphinxDirective):
    """
    Like the standard "Include" directive, but interprets absolute paths
    "correctly", i.e. relative to source directory.
    """

    def run(self) -> list[Node]:
        if self.arguments[0].startswith('<') and \
           self.arguments[0].endswith('>'):
            # docutils "standard" includes, do not do path processing
            return super().run()
        
        rel_filename, filename = self.env.relfn2path(self.arguments[0])
        self.arguments[0] = filename
        self.env.note_included(filename)
        
        # Read the included file content to emit source-read event
        # Only emit for non-literal includes (content will be parsed as RST)
        if 'literal' not in self.options:
            try:
                with open(filename, encoding=self.state.document.settings.input_encoding,
                          errors=self.state.document.settings.input_encoding_error_handler) as f:
                    raw_content = f.read()
                
                # Apply start-line/end-line filtering
                startline = self.options.get('start-line', None)
                endline = self.options.get('end-line', None)
                if startline or endline is not None:
                    lines_list = raw_content.splitlines(keepends=True)
                    raw_text = ''.join(lines_list[startline:endline])
                else:
                    raw_text = raw_content
                
                # Apply start-after/end-before filtering
                after_text = self.options.get('start-after', None)
                if after_text:
                    after_index = raw_text.find(after_text)
                    if after_index < 0:
                        raise self.severe('Problem with "start-after" option of "%s" '
                                          'directive:\\nText not found.' % self.name)
                    raw_text = raw_text[after_index + len(after_text):]
                before_text = self.options.get('end-before', None)
                if before_text:
                    before_index = raw_text.find(before_text)
                    if before_index < 0:
                        raise self.severe('Problem with "end-before" option of "%s" '
                                          'directive:\\nText not found.' % self.name)
                    raw_text = raw_text[:before_index]
                
                # Emit source-read event for the included file
                # Use the docname (without suffix) as the key
                docname = rel_filename
                for suffix in self.config.source_suffix:
                    if docname.endswith(suffix):
                        docname = docname[:-len(suffix)]
                        break
                arg = [raw_text]
                self.env.events.emit('source-read', docname, arg)
                
                # Check if the source was modified by handlers
                # If so, use the modified content
                if arg and arg[0] != raw_text:
                    raw_text = arg[0]
                
                # Store the processed content in the state so it can be reused
                # This allows handlers to modify the content and have the modification propagate
                self.state.document.settings.include_processed_text = raw_text
                
            except Exception:
                # If reading fails, fall back to default behavior
                pass
        
        # Call parent run to process the include
        # For literal includes, the parent class will handle them correctly
        return super().run()
