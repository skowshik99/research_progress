class Include(BaseInclude, SphinxDirective):
    """
    Like the standard "Include" directive, but interprets absolute paths
    "correctly", i.e. relative to source directory.
    """

    def run(self) -> list[Node]:
        if self.arguments[0].startswith('<') and \
           self.arguments[0].endswith('>'):
            # docutils "standard" includes, do not do path processing
            return super().run()
        
        rel_filename, filename = self.env.relfn2path(self.arguments[0])
        self.arguments[0] = filename
        
        # Read the file content ourselves to support source-read event
        try:
            with open(filename, encoding='utf-8', 
                     errors=self.state.document.settings.input_encoding_error_handler) as f:
                content = f.read()
        except UnicodeError:
            raise self.severe(f'Problem with "{self.name}" directive:\n'
                              f'{io.error_string(UnicodeError())}')
        
        # Determine docname from filename
        docname = self.env.path2doc(filename) if filename else None
        
        # Emit "source-read" event with docname
        arg = [content]
        self.env.events.emit('source-read', docname, arg)
        content = arg[0]
        
        # Create a string-based FileInput for docutils
        input = SphinxStringFileInput(filename, content)
        
        self.env.note_included(filename)
